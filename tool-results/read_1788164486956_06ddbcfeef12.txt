   214→interface Segment {
   215→  start: number;
   216→  end: number;
   217→}

   219→/** Split a 1-D wall run by door openings (pos is 0..1 along the wall). */
   220→function splitWallByOpenings(
   221→  wallLength: number,
   222→  openings: { pos: number; width: number }[],
   223→): Segment[] {
   224→  if (wallLength <= 0) return [];
   225→  if (openings.length === 0) return [{ start: 0, end: wallLength }];
   226→  const sorted = openings
   227→    .map((o) => ({
   228→      start: o.pos * wallLength - o.width / 2,
   229→      end: o.pos * wallLength + o.width / 2,
   230→    }))
   231→    .filter((o) => o.end > 0.05 && o.start < wallLength - 0.05)
   232→    .map((o) => ({
   233→      start: Math.max(0, o.start),
   234→      end: Math.min(wallLength, o.end),
   235→    }))
   236→    .sort((a, b) => a.start - b.start);
   237→  const segments: Segment[] = [];
   238→  let cursor = 0;
   239→  for (const o of sorted) {
   240→    if (o.start > cursor + 0.05) segments.push({ start: cursor, end: o.start });
   241→    cursor = Math.max(cursor, o.end);
   242→  }
   243→  if (cursor < wallLength - 0.05) segments.push({ start: cursor, end: wallLength });
   244→  return segments.filter((s) => s.end - s.start > 0.05);
   245→}

   247→function doorsOn(room: RoomRect, wall: 'top' | 'right' | 'bottom' | 'left') {
   248→  return room.doors.filter((d) => d.wall === wall);
   249→}
   250→function windowsOn(room: RoomRect, wall: 'top' | 'right' | 'bottom' | 'left') {
   251→  return room.windows.filter((w) => w.wall === wall);
   252→}

   254→/** Mix a hex colour toward neutral grey by `amount` (0..1) for floor realism. */
   255→function desaturate(hex: string, amount: number): string {
   256→  const c = new THREE.Color(hex);
   257→  const gray = (c.r + c.g + c.b) / 3;
   258→  c.r = c.r * (1 - amount) + gray * amount;
   259→  c.g = c.g * (1 - amount) + gray * amount;
   260→  c.b = c.b * (1 - amount) + gray * amount;
   261→  return `#${c.getHexString()}`;
   262→}

   309→
   310→function computeScene(
   311→  layout: LayoutData,
   312→  floor: number,
   313→  showAllFloors: boolean,
   314→  style: DesignStyle,
   315→  showWalls: boolean,
   316→  showFurniture: boolean,
   317→): ComputedScene {
   318→  const cfg = STYLE_CONFIG[style];
   319→  const wallHeight = cfg.wallHeight;
   320→  const plotW = layout.plot.width;
   321→  const plotL = layout.plot.length;
   322→
   323→  // Decide which floors to render.
   324→  const floorsRendered: number[] = [];
   325→  if (showAllFloors) {
   326→    for (let f = 0; f < layout.floors; f++) floorsRendered.push(f);
   327→  } else {
   328→    const f = Math.max(0, Math.min(floor, Math.max(0, layout.floors - 1)));
   329→    floorsRendered.push(f);
   330→  }
   331→  // Single-floor view always renders at ground level (baseY = 0).
   332→  // Multi-floor view stacks at idx * FLOOR_HEIGHT.
   333→  const floorBaseYs = floorsRendered.map((_, idx) =>
   334→    showAllFloors ? idx * FLOOR_HEIGHT : 0,
   335→  );
   336→
   337→  const floorSlabs: SlabData[] = [];
   338→  const interFloorSlabs: SlabData[] = [];
   339→  const walls: BoxData[] = [];
   340→  const trims: BoxData[] = [];
   341→  const windows: BoxData[] = [];
   342→  const doorPanels: BoxData[] = [];
   343→  const stairs: BoxData[] = [];
   344→  const labels: LabelData[] = [];
   345→  const furniture: FurnitureRender[] = [];
   346→
   347→  let minX = Infinity;
   348→  let maxX = -Infinity;
   349→  let minZ = Infinity;
   350→  let maxZ = -Infinity;
   351→
   352→  floorsRendered.forEach((f, idx) => {
   353→    const baseY = floorBaseYs[idx];
   354→    const floorRooms = layout.rooms.filter((r) => r.floor === f);
   355→
   356→    let fMinX = Infinity;
   357→    let fMaxX = -Infinity;
   358→    let fMinZ = Infinity;
   359→    let fMaxZ = -Infinity;
   360→
   361→    floorRooms.forEach((room) => {
   362→      const cx = to3DX(room.x + room.width / 2, plotW);
   363→      const cz = to3DZ(room.y + room.length / 2, plotL);
   364→      const isStaircase = room.type === 'staircase';
   365→      const isOpenAir = room.type === 'parking' || room.type === 'balcony';
   366→      const catalog = ROOM_CATALOG[room.type];
   367→
   368→      // Per-room coloured floor slab (slightly desaturated for realism).
   369→      const baseColor = isOpenAir ? '#cfcfcf' : catalog?.color ?? '#e8e8e8';
   370→      const slabColor = desaturate(baseColor, 0.18);
   371→      floorSlabs.push({
   372→        position: [cx, baseY + FLOOR_SLAB_THICKNESS / 2, cz],
   373→        size: [room.width, room.length],
   374→        color: slabColor,
   375→      });
   376→
   377→      // Track footprints.
   378→      const rMinX = to3DX(room.x, plotW);
   379→      const rMaxX = to3DX(room.x + room.width, plotW);
   380→      const rMinZ = to3DZ(room.y, plotL);
   381→      const rMaxZ = to3DZ(room.y + room.length, plotL);
   382→      minX = Math.min(minX, rMinX);
   383→      maxX = Math.max(maxX, rMaxX);
   384→      minZ = Math.min(minZ, rMinZ);
   385→      maxZ = Math.max(maxZ, rMaxZ);
   386→      fMinX = Math.min(fMinX, rMinX);
   387→      fMaxX = Math.max(fMaxX, rMaxX);
   388→      fMinZ = Math.min(fMinZ, rMinZ);
   389→      fMaxZ = Math.max(fMaxZ, rMaxZ);
   390→
   391→      // Walls (skip open-air rooms).
   392→      if (showWalls && !isOpenAir) {
   393→        addRoomWalls(
   394→          room,
   395→          baseY,
   396→          wallHeight,
   397→          plotW,
   398→          plotL,
   399→          walls,
   400→          trims,
   401→          windows,
   402→          doorPanels,
   403→          cfg,
   404→        );
   405→      }
   406→
   407→      // Staircase steps.
   408→      if (isStaircase) {
   409→        addStairs(room, baseY, plotW, plotL, stairs);
   410→      }
   411→
   412→      // Floating label.
   413→      labels.push({
   414→        position: [cx, baseY + wallHeight + 0.6, cz],
   415→        text: room.name,
   416→        subtext: `${room.width.toFixed(1)} \u00D7 ${room.length.toFixed(1)} ft`,
   417→        color: catalog?.accent ?? '#2b4a7a',
   418→      });
   419→    });
   420→
   421→    // Inter-floor ceiling slab between this floor and the next.
   422→    if (showAllFloors && idx < floorsRendered.length - 1 && fMinX < Infinity) {
   423→      const slabY = baseY + wallHeight + SLAB_THICKNESS / 2;
   424→      interFloorSlabs.push({
   425→        position: [(fMinX + fMaxX) / 2, slabY, (fMinZ + fMaxZ) / 2],
   426→        size: [fMaxX - fMinX, fMaxZ - fMinZ],
   427→        color: cfg.slabColor,
   428→      });
   429→    }
   430→  });
   431→
   432→  // Furniture — render via <FurnitureMesh3D>.
   433→  const allFurniture: FurnitureItem[] = layout.furniture ?? [];
   434→  if (showFurniture) {
   435→    for (const item of allFurniture) {
   436→      const idx = floorsRendered.indexOf(item.floor);
   437→      if (idx < 0) continue;
   438→      const baseY = floorBaseYs[idx];
   439→      // Furniture sits on top of the floor slab.
   440→      const worldY = baseY + FLOOR_SLAB_THICKNESS + 0.01;
   441→      const worldX = item.x + item.width / 2 - plotW / 2;
   442→      const worldZ = item.y + item.length / 2 - plotL / 2;
   443→      furniture.push({ item, worldX, worldZ, worldY });
   444→    }
   445→  }
   446→
   447→  const footprint: Footprint | null =
   448→    minX < Infinity ? { minX, maxX, minZ, maxZ } : null;
   449→  const lastBaseY = floorBaseYs[floorBaseYs.length - 1] ?? 0;
   450→  const totalBuildingHeight = lastBaseY + wallHeight;
   451→  const roofY = lastBaseY + wallHeight + 0.15;
   452→
   453→  return {
   454→    floorSlabs,
   455→    interFloorSlabs,
   456→    walls,
   457→    trims,
   458→    windows,
   459→    doorPanels,
   460→    stairs,
   461→    labels,
   462→    furniture,
   463→    footprint,
   464→    totalBuildingHeight,
   465→    roofY,
   466→  };
   467→}

   558→interface AxisWallArgs {
   559→  axis: 'x' | 'z';
   560→  fixedCoord: number;
   561→  startCoord: number;
   562→  length: number;
   563→  doors: DoorMarker[];
   564→  wins: WindowMarker[];
   565→  baseY: number;
   566→  wallHeight: number;
   567→  wallCenterY: number;
   568→  walls: BoxData[];
   569→  trims: BoxData[];
   570→  windows: BoxData[];
   571→  doorPanels: BoxData[];
   572→  cfg: StyleConfig;
   573→}
   574→
   575→function buildAxisWall(a: AxisWallArgs) {
   576→  const segments = splitWallByOpenings(
   577→    a.length,
   578→    a.doors.map((d) => ({ pos: d.pos, width: d.width })),
   579→  );
   580→  const isX = a.axis === 'x';
   581→
   582→  // Solid wall segments (with door gaps).
   583→  for (const seg of segments) {
   584→    const segStartWorld = a.startCoord + seg.start;
   585→    const segEndWorld = a.startCoord + seg.end;
   586→    const center = (segStartWorld + segEndWorld) / 2;
   587→    const span = segEndWorld - segStartWorld;
   588→    if (span <= 0.05) continue;
   589→
   590→    if (isX) {
   591→      a.walls.push({
   592→        position: [center, a.wallCenterY, a.fixedCoord],
   593→        size: [span, a.wallHeight, WALL_THICKNESS],
   594→      });
   595→      if (a.cfg.accentTrim) {
   596→        a.trims.push({
   597→          position: [center, a.baseY + a.wallHeight + 0.1, a.fixedCoord],
   598→          size: [span, 0.2, WALL_THICKNESS * 1.05],
   599→        });
   600→      }
   601→    } else {
   602→      a.walls.push({
   603→        position: [a.fixedCoord, a.wallCenterY, center],
   604→        size: [WALL_THICKNESS, a.wallHeight, span],
   605→      });
   606→      if (a.cfg.accentTrim) {
   607→        a.trims.push({
   608→          position: [a.fixedCoord, a.baseY + a.wallHeight + 0.1, center],
   609→          size: [WALL_THICKNESS * 1.05, 0.2, span],
   610→        });
   611→      }
   612→    }
   613→  }
   614→
   615→  // Windows — translucent cyan overlays on the wall surface.
   616→  for (const win of a.wins) {
   617→    const wWidth = Math.min(win.width * a.cfg.windowScale, a.length * 0.7);
   618→    const center = a.startCoord + win.pos * a.length;
   619→    const winY = a.baseY + WINDOW_BOTTOM + WINDOW_HEIGHT / 2;
   620→    const thick = WALL_THICKNESS * 1.4;
   621→    if (isX) {
   622→      a.windows.push({
   623→        position: [center, winY, a.fixedCoord],
   624→        size: [wWidth, WINDOW_HEIGHT, thick],
   625→      });
   626→    } else {
   627→      a.windows.push({
   628→        position: [a.fixedCoord, winY, center],
   629→        size: [thick, WINDOW_HEIGHT, wWidth],
   630→      });
   631→    }
   632→  }
   633→
   634→  // Door panels — thin leaf at the door position.
   635→  for (const d of a.doors) {
   636→    const center = a.startCoord + d.pos * a.length;
   637→    const panelThick = WALL_THICKNESS * 0.55;
   638→    if (isX) {
   639→      a.doorPanels.push({
   640→        position: [center, a.baseY + DOOR_HEIGHT / 2, a.fixedCoord],
   641→        size: [d.width * 0.92, DOOR_HEIGHT, panelThick],
   642→      });
   643→    } else {
   644→      a.doorPanels.push({
   645→        position: [a.fixedCoord, a.baseY + DOOR_HEIGHT / 2, center],
   646→        size: [panelThick, DOOR_HEIGHT, d.width * 0.92],
   647→      });
   648→    }
   649→  }

   682→interface RoofProps {
   683→  footprint: Footprint;
   684→  roofY: number;
   685→  cfg: StyleConfig;
   686→  accentColor: string;

   689→function Roof({
   690→  footprint,
   691→  roofY,
   692→  cfg,
   693→  accentColor,
   694→}: RoofProps): React.JSX.Element {
   695→  const { minX, maxX, minZ, maxZ } = footprint;
   696→  const cx = (minX + maxX) / 2;
   697→  const cz = (minZ + maxZ) / 2;
   698→  const w = maxX - minX;
   699→  const d = maxZ - minZ;
   700→  const isPitched = cfg.roofType === 'pitched';
   701→  const overhang = cfg.roofOverhang;
   702→  const totalW = w + overhang * 2;
   703→  const totalD = d + overhang * 2;
   704→  const pitchH = totalW * PITCH_RATIO;
   705→
   706→  // Triangular-prism ExtrudeGeometry for the pitched roof.
   707→  const geom = useMemo(() => {
   708→    if (!isPitched) return null;
   709→    const shape = new THREE.Shape();
   710→    shape.moveTo(-totalW / 2, 0);
   711→    shape.lineTo(totalW / 2, 0);
   712→    shape.lineTo(0, pitchH);
   713→    shape.closePath();
   714→    const g = new THREE.ExtrudeGeometry(shape, {
   715→      depth: totalD,
   716→      bevelEnabled: false,
   717→    });
   718→    g.translate(0, 0, -totalD / 2);
   719→    g.computeVertexNormals();
   720→    return g;
   721→  }, [isPitched, totalW, pitchH, totalD]);
   722→
   723→  if (isPitched && geom) {
   724→    return (
   725→      <mesh geometry={geom} position={[cx, roofY, cz]} castShadow receiveShadow>
   726→        <meshStandardMaterial
   727→          color={cfg.pitchedRoofColor}
   728→          roughness={0.7}
   729→          metalness={0.05}
   730→        />
   731→      </mesh>
   732→    );
   733→  }
   734→
   735→  // Flat slab (optionally with overhang) — semi-transparent accent colour.
   736→  return (
   737→    <mesh
   738→      position={[cx, roofY + SLAB_THICKNESS / 2, cz]}
   739→      castShadow
   740→      receiveShadow
   741→    >
   742→      <boxGeometry args={[totalW, SLAB_THICKNESS, totalD]} />
   743→      <meshStandardMaterial
   744→        color={accentColor}
   745→        roughness={0.5}
   746→        metalness={0.1}
   747→        transparent
   748→        opacity={cfg.roofOpacity}
   749→      />
   750→    </mesh>
   751→  );
   752→}

   868→// -------------------------------------------------------------------------
   869→// CameraRig — animated camera transitions on cameraView change.
   870→// Cancels animation when the user starts dragging OrbitControls.
   871→// -------------------------------------------------------------------------
   872→
   873→type ControlsLike = {
   874→  target: THREE.Vector3;
   875→  update: () => void;
   876→  addEventListener: (type: string, listener: () => void) => void;
   877→  removeEventListener: (type: string, listener: () => void) => void;
   878→};
   879→
   880→interface CameraRigProps {
   881→  cameraView: 'orbit' | 'top' | 'front' | 'isometric';
   882→  plotW: number;
   883→  plotL: number;
   884→}
   885→
   886→function CameraRig({
   887→  cameraView,
   888→  plotW,
   889→  plotL,
   890→}: CameraRigProps): React.JSX.Element | null {
   891→  const cameraTargetRef = useRef(new THREE.Vector3());
   892→  const lookAtRef = useRef(new THREE.Vector3());
   893→  const animatingRef = useRef(true);
   894→
   895→  const camera = useThree((s) => s.camera);
   896→  const controls = useThree((s) => s.controls) as unknown as ControlsLike | null;
   897→
   898→  // Compute target position + lookAt whenever cameraView changes.
   899→  useEffect(() => {
   900→    let pos: [number, number, number];
   901→    let look: [number, number, number];
   902→    switch (cameraView) {
   903→      case 'isometric':
   904→        pos = [plotW * 0.7, plotW * 0.8, plotW * 0.7];
   905→        look = [0, 4, 0];
   906→        break;
   907→      case 'front':
   908→        pos = [0, 6, plotL * 1.1];
   909→        look = [0, 5, 0];
   910→        break;
   911→      case 'top':
   912→        pos = [0, plotL * 1.5, 0.01];
   913→        look = [0, 0, 0];
   914→        break;
   915→      case 'orbit':
   916→      default:
   917→        pos = [plotW * 0.8, plotL * 0.7, plotW * 0.9];
   918→        look = [0, 4, 0];
   919→        break;
   920→    }
   921→    cameraTargetRef.current.set(pos[0], pos[1], pos[2]);
   922→    lookAtRef.current.set(look[0], look[1], look[2]);
   923→    animatingRef.current = true;
   924→  }, [cameraView, plotW, plotL]);
   925→
   926→  // Cancel animation on user interaction with OrbitControls.
   927→  useEffect(() => {
   928→    if (!controls) return;
   929→    const onStart = () => {
   930→      animatingRef.current = false;
   931→    };
   932→    controls.addEventListener('start', onStart);
   933→    return () => controls.removeEventListener('start', onStart);
   934→  }, [controls]);
   935→
   936→  useFrame(() => {
   937→    if (!animatingRef.current) return;
   938→    camera.position.lerp(cameraTargetRef.current, 0.08);
   939→    if (controls) {
   940→      controls.target.lerp(lookAtRef.current, 0.08);
   941→      controls.update();
   942→    } else {
   943→      camera.lookAt(lookAtRef.current);
   944→    }
   945→    if (camera.position.distanceTo(cameraTargetRef.current) < 0.15) {
   946→      animatingRef.current = false;
   947→    }
   948→  });
   949→
   950→  return null;
   951→}

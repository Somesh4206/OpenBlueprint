     3→     3→// =========================================================================
     4→     4→// OpenBlueprint — 3D Furniture Models Library (Task 5-a)
     5→     5→// React Three Fiber component that renders a recognizable 3D model for
     6→     6→// each FurnitureType, built from Three.js primitives (boxes, cylinders,
     7→     7→// spheres, cones, torus).
     8→     8→//
     9→     9→// Coordinate convention (matches viewer-3d.tsx):
    10→    10→//   - 1 unit = 1 foot
    11→    11→//   - Origin at the centre of the plot, +Y is up
    12→    12→//   - The parent viewer passes the ALREADY-CONVERTED world-space centre
    13→    13→//     of the furniture via `worldX` / `worldZ`.
    14→    14→//   - The model is built in local space (centred on origin, sitting on
    15→    15→//     the floor at y=0), then placed by a single parent <group> with
    16→    16→//     position=[worldX, 0, worldZ] and rotation=[0, item.rotation*π/180, 0].
    17→    17→//
    18→    18→// Each model scales to fit `item.width` (local X) × `item.length`
    19→    19→// (local Z). Local +Z is treated as the "back" of the furniture
    20→    20→// (headboard, sofa backrest, chair backrest, etc.).
    21→    21→// =========================================================================
    22→    22→
    23→    23→import React, { useMemo } from 'react';
    24→    24→import type { FurnitureItem, FurnitureType } from '@/lib/types';
    25→    25→
    26→    26→// -------------------------------------------------------------------------
    27→    27→// Public API
    28→    28→// -------------------------------------------------------------------------
    29→    29→
    30→    30→export interface FurnitureMesh3DProps {
    31→    31→  item: FurnitureItem;
    32→    32→  /** Already-converted 3D X coordinate (feet, plot-centred). */
    33→    33→  worldX: number;
    34→    34→  /** Already-converted 3D Z coordinate (feet, plot-centred). */
    35→    35→  worldZ: number;
    36→    36→}
    37→    37→
    38→    38→export function FurnitureMesh3D(props: FurnitureMesh3DProps): React.JSX.Element {
    39→    39→  const { item, worldX, worldZ } = props;
    40→    40→  const color = item.color || '#9aa0a6';
    41→    41→  const w = Math.max(item.width, 0.5);
    42→    42→  const l = Math.max(item.length, 0.5);
    43→    43→
    44→    44→  // Memoize the model subtree so changing position/rotation on the parent
    45→    45→  // group does not rebuild the entire mesh tree.
    46→    46→  const model = useMemo(
    47→    47→    () => buildFurnitureModel(item.type, w, l, color),
    48→    48→    [item.type, w, l, color],
    49→    49→  );
    50→    50→
    51→    51→  const rotY = ((item.rotation || 0) * Math.PI) / 180;
    52→    52→
    53→    53→  return (
    54→    54→    <group position={[worldX, 0, worldZ]} rotation={[0, rotY, 0]}>
    55→    55→      {model}
    56→    56→    </group>
    57→    57→  );
    58→    58→}

    66→    66→/** Darken a hex colour by `amount` (0..1). Returns the original on parse error. */
    67→    67→function darken(hex: string, amount: number): string {
    68→    68→  const m = /^#?([0-9a-f]{6})$/i.exec(hex);
    69→    69→  if (!m) return hex;
    70→    70→  const num = parseInt(m[1], 16);
    71→    71→  let r = (num >> 16) & 0xff;
    72→    72→  let g = (num >> 8) & 0xff;
    73→    73→  let b = num & 0xff;
    74→    74→  const f = Math.max(0, 1 - amount);
    75→    75→  r = Math.min(255, Math.round(r * f));
    76→    76→  g = Math.min(255, Math.round(g * f));
    77→    77→  b = Math.min(255, Math.round(b * f));
    78→    78→  return '#' + ((1 << 24) + (r << 16) + (g << 8) + b).toString(16).slice(1);
    79→    79→}
    80→    80→
    81→    81→/** Lighten a hex colour by `amount` (0..1) toward white. */
    82→    82→function lighten(hex: string, amount: number): string {
    83→    83→  const m = /^#?([0-9a-f]{6})$/i.exec(hex);
    84→    84→  if (!m) return hex;
    85→    85→  const num = parseInt(m[1], 16);
    86→    86→  let r = (num >> 16) & 0xff;
    87→    87→  let g = (num >> 8) & 0xff;
    88→    88→  let b = num & 0xff;
    89→    89→  r = Math.min(255, Math.round(r + (255 - r) * amount));
    90→    90→  g = Math.min(255, Math.round(g + (255 - g) * amount));
    91→    91→  b = Math.min(255, Math.round(b + (255 - b) * amount));
    92→    92→  return '#' + ((1 << 24) + (r << 16) + (g << 8) + b).toString(16).slice(1);
    93→    93→}

  1086→  1086→function darkenHex(hex: string, amount: number): string {
  1087→  1087→  const h = hex.replace('#', '');

  1089→  1089→  const g = parseInt(h.slice(2, 4), 16);
  1090→  1090→  const b = parseInt(h.slice(4, 6), 16);
  1091→  1091→  const dr = Math.round(r * (1 - amount));
  1092→  1092→  const dg = Math.round(g * (1 - amount));
  1093→  1093→  const db = Math.round(b * (1 - amount));
  1094→  1094→  return '#' + [dr, dg, db].map((x) => x.toString(16).padStart(2, '0')).join('');

     3→// =========================================================================
     4→// OpenBlueprint — 3D Furniture Models Library (Task 5-a)
     5→// React Three Fiber component that renders a recognizable 3D model for
     6→// each FurnitureType, built from Three.js primitives (boxes, cylinders,
     7→// spheres, cones, torus).
     8→//
     9→// Coordinate convention (matches viewer-3d.tsx):
    10→//   - 1 unit = 1 foot
    11→//   - Origin at the centre of the plot, +Y is up
    12→//   - The parent viewer passes the ALREADY-CONVERTED world-space centre
    13→//     of the furniture via `worldX` / `worldZ`.
    14→//   - The model is built in local space (centred on origin, sitting on
    15→//     the floor at y=0), then placed by a single parent <group> with
    16→//     position=[worldX, 0, worldZ] and rotation=[0, item.rotation*π/180, 0].
    17→//
    18→// Each model scales to fit `item.width` (local X) × `item.length`
    19→// (local Z). Local +Z is treated as the "back" of the furniture
    20→// (headboard, sofa backrest, chair backrest, etc.).
    21→// =========================================================================
    22→
    23→import React, { useMemo } from 'react';
    24→import type { FurnitureItem, FurnitureType } from '@/lib/types';
    25→
    26→// -------------------------------------------------------------------------
    27→// Public API
    28→// -------------------------------------------------------------------------
    29→
    30→export interface FurnitureMesh3DProps {
    31→  item: FurnitureItem;
    32→  /** Already-converted 3D X coordinate (feet, plot-centred). */
    33→  worldX: number;
    34→  /** Already-converted 3D Z coordinate (feet, plot-centred). */
    35→  worldZ: number;
    36→}
    37→
    38→export function FurnitureMesh3D(props: FurnitureMesh3DProps): React.JSX.Element {
    39→  const { item, worldX, worldZ } = props;
    40→  const color = item.color || '#9aa0a6';
    41→  const w = Math.max(item.width, 0.5);
    42→  const l = Math.max(item.length, 0.5);
    43→
    44→  // Memoize the model subtree so changing position/rotation on the parent
    45→  // group does not rebuild the entire mesh tree.
    46→  const model = useMemo(
    47→    () => buildFurnitureModel(item.type, w, l, color),
    48→    [item.type, w, l, color],
    49→  );
    50→
    51→  const rotY = ((item.rotation || 0) * Math.PI) / 180;
    52→
    53→  return (
    54→    <group position={[worldX, 0, worldZ]} rotation={[0, rotY, 0]}>
    55→      {model}
    56→    </group>
    57→  );
    58→}
    59→
    60→export default FurnitureMesh3D;

    66→/** Darken a hex colour by `amount` (0..1). Returns the original on parse error. */
    67→function darken(hex: string, amount: number): string {
    68→  const m = /^#?([0-9a-f]{6})$/i.exec(hex);
    69→  if (!m) return hex;
    70→  const num = parseInt(m[1], 16);
    71→  let r = (num >> 16) & 0xff;
    72→  let g = (num >> 8) & 0xff;
    73→  let b = num & 0xff;
    74→  const f = Math.max(0, 1 - amount);
    75→  r = Math.min(255, Math.round(r * f));
    76→  g = Math.min(255, Math.round(g * f));
    77→  b = Math.min(255, Math.round(b * f));
    78→  return '#' + ((1 << 24) + (r << 16) + (g << 8) + b).toString(16).slice(1);
    79→}
    80→
    81→/** Lighten a hex colour by `amount` (0..1) toward white. */
    82→function lighten(hex: string, amount: number): string {
    83→  const m = /^#?([0-9a-f]{6})$/i.exec(hex);
    84→  if (!m) return hex;
    85→  const num = parseInt(m[1], 16);
    86→  let r = (num >> 16) & 0xff;
    87→  let g = (num >> 8) & 0xff;
    88→  let b = num & 0xff;
    89→  r = Math.min(255, Math.round(r + (255 - r) * amount));
    90→  g = Math.min(255, Math.round(g + (255 - g) * amount));
    91→  b = Math.min(255, Math.round(b + (255 - b) * amount));
    92→  return '#' + ((1 << 24) + (r << 16) + (g << 8) + b).toString(16).slice(1);
    93→}

   110→function Box({
   111→  position = [0, 0, 0],
   112→  size,
   113→  color,
   114→  roughness = 0.7,
   115→  metalness = 0,

   123→        color={color}
   124→        roughness={roughness}
   125→        metalness={metalness}
   126→        transparent={opacity < 1}
   127→        opacity={opacity}
   128→      />
   129→    </mesh>
   130→  );

   822→  const tableW = w * 0.6;
   823→  const tableL = l * 0.5;
   824→  const chairColor = darken(color, 0.45);
   825→  // Chair layout: chairs/2 along each long side (and 1 on each short side if 6)
   826→  const layout: Array<[number, number, number]> = [];
   827→  if (chairs === 4) {
   828→    layout.push([-w * 0.15, 0, -l * 0.4]);
   829→    layout.push([w * 0.15, 0, -l * 0.4]);
   830→    layout.push([-w * 0.15, 0, l * 0.4]);
   831→    layout.push([w * 0.15, 0, l * 0.4]);
   832→  } else {
   833→    layout.push([-w * 0.22, 0, -l * 0.4]);
   834→    layout.push([0, 0, -l * 0.4]);
   835→    layout.push([w * 0.22, 0, -l * 0.4]);
   836→    layout.push([-w * 0.22, 0, l * 0.4]);
   837→    layout.push([0, 0, l * 0.4]);
   838→    layout.push([w * 0.22, 0, l * 0.4]);
   839→  }
   840→  return (
   841→    <group>
   842→      {/* Table (rect, dining height) */}
   843→      <TableRectModel w={tableW} l={tableL} color={color} height={2.5} />
   844→      {/* Chairs (rotated to face the table) */}
   845→      {layout.map(([cx, cy, cz], i) => {
   846→        const isTop = cz < 0;
   847→        return (
   848→          <group key={i} position={[cx, cy, cz]} rotation={[0, isTop ? Math.PI : 0, 0]}>
   849→            <ChairDiningModel w={Math.min(w, l) * 0.22} l={Math.min(w, l) * 0.22} color={chairColor} />
   850→          </group>
   851→        );
   852→      })}
   853→    </group>
   854→  );
   855→}

  1086→function darkenHex(hex: string, amount: number): string {
  1087→  const h = hex.replace('#', '');
  1088→  const r = parseInt(h.slice(0, 2), 16);
  1089→  const g = parseInt(h.slice(2, 4), 16);
  1090→  const b = parseInt(h.slice(4, 6), 16);
  1091→  const dr = Math.round(r * (1 - amount));
  1092→  const dg = Math.round(g * (1 - amount));
  1093→  const db = Math.round(b * (1 - amount));
  1094→  return '#' + [dr, dg, db].map((x) => x.toString(16).padStart(2, '0')).join('');
  1095→}

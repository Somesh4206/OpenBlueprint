   188→function to3DX(x: number, plotWidth: number): number {
   189→  return x - plotWidth / 2;

   192→  return y - plotLength / 2;

   195→interface Segment {
   196→  start: number;
   197→  end: number;
   198→}
   199→
   200→/** Split a 1-D wall run by door openings (pos is 0..1 along the wall). */
   201→function splitWallByOpenings(
   202→  wallLength: number,
   203→  openings: { pos: number; width: number }[],
   204→): Segment[] {
   205→  if (wallLength <= 0) return [];
   206→  if (openings.length === 0) return [{ start: 0, end: wallLength }];
   207→  const sorted = openings
   208→    .map((o) => ({
   209→      start: o.pos * wallLength - o.width / 2,
   210→      end: o.pos * wallLength + o.width / 2,
   211→    }))
   212→    .filter((o) => o.end > 0.05 && o.start < wallLength - 0.05)
   213→    .map((o) => ({
   214→      start: Math.max(0, o.start),
   215→      end: Math.min(wallLength, o.end),
   216→    }))
   217→    .sort((a, b) => a.start - b.start);
   218→  const segments: Segment[] = [];
   219→  let cursor = 0;
   220→  for (const o of sorted) {
   221→    if (o.start > cursor + 0.05) segments.push({ start: cursor, end: o.start });
   222→    cursor = Math.max(cursor, o.end);
   223→  }
   224→  if (cursor < wallLength - 0.05) segments.push({ start: cursor, end: wallLength });
   225→  return segments.filter((s) => s.end - s.start > 0.05);
   226→}

   232→  return room.windows.filter((w) => w.wall === wall);

   260→interface ComputedScene {
   261→  floorSlabs: SlabData[];
   262→  interFloorSlabs: SlabData[];
   263→  walls: BoxData[];

   265→  windows: BoxData[];

   274→
   275→function computeScene(
   276→  layout: LayoutData,
   277→  floor: number,
   278→  showAllFloors: boolean,
   279→  style: DesignStyle,
   280→  showWalls: boolean,
   281→  showFurniture: boolean,
   282→): ComputedScene {
   283→  const cfg = STYLE_CONFIG[style];
   284→  const wallHeight = cfg.wallHeight;
   285→  const floorHeight = wallHeight + cfg.slabThickness;
   286→  const plotW = layout.plot.width;
   287→  const plotL = layout.plot.length;
   288→
   289→  const floorsRendered: number[] = [];
   290→  if (showAllFloors) {
   291→    for (let f = 0; f < layout.floors; f++) floorsRendered.push(f);
   292→  } else {
   293→    // Always include the requested floor (clamp to valid range).
   294→    const f = Math.max(0, Math.min(floor, Math.max(0, layout.floors - 1)));
   295→    floorsRendered.push(f);
   296→  }
   297→  const floorBaseYs = floorsRendered.map((f) => f * floorHeight);
   298→
   299→  const floorSlabs: SlabData[] = [];
   300→  const interFloorSlabs: SlabData[] = [];
   301→  const walls: BoxData[] = [];
   302→  const trims: BoxData[] = [];
   303→  const windows: BoxData[] = [];
   304→  const doorPanels: BoxData[] = [];
   305→  const stairs: BoxData[] = [];
   306→  const furniture: BoxData[] = [];
   307→  const labels: LabelData[] = [];
   308→
   309→  let minX = Infinity;
   310→  let maxX = -Infinity;
   311→  let minZ = Infinity;
   312→  let maxZ = -Infinity;
   313→
   314→  floorsRendered.forEach((f, idx) => {
   315→    const baseY = floorBaseYs[idx];
   316→    const floorRooms = layout.rooms.filter((r) => r.floor === f);
   317→
   318→    // Footprint of this floor (for inter-floor slab).
   319→    let fMinX = Infinity;
   320→    let fMaxX = -Infinity;
   321→    let fMinZ = Infinity;
   322→    let fMaxZ = -Infinity;
   323→
   324→    floorRooms.forEach((room) => {
   325→      const cx = to3DX(room.x + room.width / 2, plotW);
   326→      const cz = to3DZ(room.y + room.length / 2, plotL);
   327→      const isStaircase = room.type === 'staircase';
   328→      const isOpenAir = room.type === 'parking' || room.type === 'balcony';
   329→      const catalog = ROOM_CATALOG[room.type];
   330→
   331→      // Floor slab (per-room, colored by catalog color).
   332→      const slabColor = isOpenAir ? '#cfcfcf' : catalog?.color ?? '#e8e8e8';
   333→      floorSlabs.push({
   334→        position: [cx, baseY + 0.02, cz],
   335→        size: [room.width, room.length],
   336→        color: slabColor,
   337→      });
   338→
   339→      // Track footprints.
   340→      const rMinX = to3DX(room.x, plotW);
   341→      const rMaxX = to3DX(room.x + room.width, plotW);
   342→      const rMinZ = to3DZ(room.y, plotL);
   343→      const rMaxZ = to3DZ(room.y + room.length, plotL);
   344→      minX = Math.min(minX, rMinX);
   345→      maxX = Math.max(maxX, rMaxX);
   346→      minZ = Math.min(minZ, rMinZ);
   347→      maxZ = Math.max(maxZ, rMaxZ);
   348→      fMinX = Math.min(fMinX, rMinX);
   349→      fMaxX = Math.max(fMaxX, rMaxX);
   350→      fMinZ = Math.min(fMinZ, rMinZ);
   351→      fMaxZ = Math.max(fMaxZ, rMaxZ);
   352→
   353→      // Walls (skip open-air rooms like parking/balcony).
   354→      if (showWalls && !isOpenAir) {
   355→        addRoomWalls(
   356→          room,
   357→          baseY,
   358→          wallHeight,
   359→          plotW,
   360→          plotL,
   361→          walls,
   362→          trims,
   363→          windows,
   364→          doorPanels,
   365→          cfg,
   366→          isStaircase,
   367→        );
   368→      }
   369→
   370→      // Staircase steps.
   371→      if (isStaircase) {
   372→        addStairs(room, baseY, plotW, plotL, stairs);
   373→      }
   374→
   375→      // Furniture.
   376→      if (showFurniture && !isStaircase && !isOpenAir) {
   377→        addFurniture(room, baseY, plotW, plotL, furniture);
   378→      }
   379→
   380→      // Floating label.
   381→      labels.push({
   382→        position: [cx, baseY + wallHeight + 0.6, cz],
   383→        text: room.name,
   384→        subtext: `${room.width.toFixed(1)}\u00A0\u00D7\u00A0${room.length.toFixed(1)}\u00A0ft`,
   385→        color: catalog?.accent ?? '#2b4a7a',
   386→      });
   387→    });
   388→
   389→    // Inter-floor slab (only between this floor and the next).
   390→    if (showAllFloors && idx < floorsRendered.length - 1 && fMinX < Infinity) {
   391→      const slabY = baseY + wallHeight + cfg.slabThickness / 2;
   392→      interFloorSlabs.push({
   393→        position: [(fMinX + fMaxX) / 2, slabY, (fMinZ + fMaxZ) / 2],
   394→        size: [fMaxX - fMinX, fMaxZ - fMinZ],
   395→        color: cfg.slabColor,
   396→      });
   397→    }
   398→  });
   399→
   400→  const footprint: Footprint | null =
   401→    minX < Infinity ? { minX, maxX, minZ, maxZ } : null;
   402→  const lastBaseY = floorBaseYs[floorBaseYs.length - 1] ?? 0;
   403→  const totalBuildingHeight = lastBaseY + wallHeight;
   404→  const roofY = lastBaseY + wallHeight + 0.15;
   405→
   406→  return {
   407→    floorSlabs,
   408→    interFloorSlabs,
   409→    walls,
   410→    trims,
   411→    windows,
   412→    doorPanels,
   413→    stairs,
   414→    furniture,
   415→    labels,
   416→    footprint,
   417→    totalBuildingHeight,
   418→    roofY,
   419→  };
   420→}

   427→function addRoomWalls(
   428→  room: RoomRect,

   483→  buildAxisWall({
   484→    axis: 'z',
   485→    fixedCoord: to3DX(room.x, plotW),
   486→    startCoord: to3DZ(room.y, plotL),
   487→    length: room.length,
   488→    doors: doorsOn(room, 'left'),
   489→    wins: windowsOn(room, 'left'),
   490→    baseY,
   491→    wallHeight,
   492→    wallCenterY,
   493→    walls,
   494→    trims,
   495→    windows,
   496→    doorPanels,
   497→    cfg,
   498→  });
   499→
   500→  // ---- RIGHT wall: along Z, at x = to3DX(room.x + room.width) ----
   501→  buildAxisWall({
   502→    axis: 'z',
   503→    fixedCoord: to3DX(room.x + room.width, plotW),
   504→    startCoord: to3DZ(room.y, plotL),
   505→    length: room.length,
   506→    doors: doorsOn(room, 'right'),
   507→    wins: windowsOn(room, 'right'),
   508→    baseY,
   509→    wallHeight,
   510→    wallCenterY,
   511→    walls,
   512→    trims,
   513→    windows,
   514→    doorPanels,
   515→    cfg,
   516→  });

   519→interface AxisWallArgs {
   520→  axis: 'x' | 'z';
   521→  fixedCoord: number;
   522→  startCoord: number;
   523→  length: number;
   524→  doors: DoorMarker[];
   525→  wins: WindowMarker[];
   526→  baseY: number;
   527→  wallHeight: number;
   528→  wallCenterY: number;
   529→  walls: BoxData[];
   530→  trims: BoxData[];
   531→  windows: BoxData[];
   532→  doorPanels: BoxData[];
   533→  cfg: StyleConfig;
   534→}
   535→
   536→function buildAxisWall(a: AxisWallArgs) {
   537→  const segments = splitWallByOpenings(
   538→    a.length,
   539→    a.doors.map((d) => ({ pos: d.pos, width: d.width })),
   540→  );
   541→  const isX = a.axis === 'x';
   542→
   543→  for (const seg of segments) {
   544→    const segStartWorld = a.startCoord + seg.start;
   545→    const segEndWorld = a.startCoord + seg.end;
   546→    const center = (segStartWorld + segEndWorld) / 2;
   547→    const span = segEndWorld - segStartWorld;
   548→    if (span <= 0.05) continue;
   549→
   550→    if (isX) {
   551→      a.walls.push({
   552→        position: [center, a.wallCenterY, a.fixedCoord],
   553→        size: [span, a.wallHeight, WALL_THICKNESS],
   554→      });
   555→      // Top accent trim
   556→      if (a.cfg.accentTrim) {
   557→        a.trims.push({
   558→          position: [center, a.baseY + a.wallHeight + 0.1, a.fixedCoord],
   559→          size: [span, 0.2, WALL_THICKNESS * 1.05],
   560→        });
   561→      }
   562→    } else {
   563→      a.walls.push({
   564→        position: [a.fixedCoord, a.wallCenterY, center],
   565→        size: [WALL_THICKNESS, a.wallHeight, span],
   566→      });
   567→      if (a.cfg.accentTrim) {
   568→        a.trims.push({
   569→          position: [a.fixedCoord, a.baseY + a.wallHeight + 0.1, center],
   570→          size: [WALL_THICKNESS * 1.05, 0.2, span],
   571→        });
   572→      }
   573→    }
   574→  }
   575→
   576→  // Windows (translucent overlays on the wall surface).
   577→  for (const win of a.wins) {
   578→    const wWidth = Math.min(win.width * a.cfg.windowScale, a.length * 0.7);
   579→    const center = a.startCoord + win.pos * a.length;
   580→    const winHeight = WINDOW_BASE_HEIGHT * a.cfg.windowScale;
   581→    const winY = a.baseY + WINDOW_BOTTOM + winHeight / 2;
   582→    const thick = WALL_THICKNESS * 1.5;
   583→    if (isX) {
   584→      a.windows.push({
   585→        position: [center, winY, a.fixedCoord],
   586→        size: [wWidth, winHeight, thick],
   587→      });
   588→    } else {
   589→      a.windows.push({
   590→        position: [a.fixedCoord, winY, center],
   591→        size: [thick, winHeight, wWidth],
   592→      });
   593→    }
   594→  }
   595→
   596→  // Door panels (thin door leaf at the door position).
   597→  for (const d of a.doors) {
   598→    const center = a.startCoord + d.pos * a.length;
   599→    const panelThick = WALL_THICKNESS * 0.55;
   600→    if (isX) {
   601→      a.doorPanels.push({
   602→        position: [center, a.baseY + DOOR_HEIGHT / 2, a.fixedCoord],
   603→        size: [d.width * 0.92, DOOR_HEIGHT, panelThick],
   604→      });
   605→    } else {
   606→      a.doorPanels.push({

   792→function Roof({
   793→  footprint,
   794→  y,
   795→  style,
   796→  accentColor,
   797→}: {
   798→  footprint: Footprint;
   799→  y: number;
   800→  style: DesignStyle;
   801→  accentColor: string;
   802→}) {
   803→  const cfg = STYLE_CONFIG[style];
   804→  const width = footprint.maxX - footprint.minX;
   805→  const depth = footprint.maxZ - footprint.minZ;
   806→  const cx = (footprint.minX + footprint.maxX) / 2;
   807→  const cz = (footprint.minZ + footprint.maxZ) / 2;
   808→  const overhang = cfg.roofOverhang;
   809→  const opacity = cfg.roofOpacity;
   810→
   811→  const pitchedGeom = useMemo(() => {
   812→    if (cfg.roofType !== 'pitched') return null;
   813→    const base = width + overhang * 2;
   814→    const apex = Math.min(width, depth) * 0.4;
   815→    const shape = new THREE.Shape();
   816→    shape.moveTo(-base / 2, 0);
   817→    shape.lineTo(base / 2, 0);
   818→    shape.lineTo(0, apex);
   819→    shape.closePath();
   820→    const extrudeDepth = depth + overhang * 2;
   821→    const g = new THREE.ExtrudeGeometry(shape, {
   822→      depth: extrudeDepth,
   823→      bevelEnabled: false,
   824→    });
   825→    g.translate(0, 0, -extrudeDepth / 2);
   826→    g.computeVertexNormals();
   827→    return g;
   828→  }, [cfg.roofType, width, depth, overhang]);
   829→
   830→  if (cfg.roofType === 'pitched' && pitchedGeom) {
   831→    return (
   832→      <mesh geometry={pitchedGeom} position={[cx, y, cz]} castShadow receiveShadow>
   833→        <meshStandardMaterial
   834→          color={accentColor}
   835→          transparent
   836→          opacity={opacity}
   837→          roughness={0.7}
   838→          side={THREE.DoubleSide}
   839→        />
   840→      </mesh>
   841→    );
   842→  }
   843→
   844→  // Flat or overhang roof — thin slab.
   845→  return (
   846→    <mesh position={[cx, y, cz]} castShadow receiveShadow>
   847→      <boxGeometry args={[width + overhang * 2, 0.3, depth + overhang * 2]} />
   848→      <meshStandardMaterial
   849→        color={accentColor}
   850→        transparent
   851→        opacity={opacity}
   852→        roughness={0.6}
   853→      />
   854→    </mesh>
   855→  );
   856→}

  1143→// =========================================================================
  1144→// Camera rig — animates the camera (and OrbitControls target) when
  1145→// `cameraView` changes. The animation auto-stops after a fixed number of
  1146→// frames OR when the user starts interacting with the controls.
  1147→// =========================================================================
  1148→
  1149→function CameraRig({
  1150→  cameraView,
  1151→  plotWidth,
  1152→  plotLength,
  1153→  buildingHeight,
  1154→}: {
  1155→  cameraView: 'orbit' | 'top' | 'front' | 'isometric';
  1156→  plotWidth: number;
  1157→  plotLength: number;
  1158→  buildingHeight: number;
  1159→}) {
  1160→  const { camera, controls } = useThree();
  1161→  const targetPos = useRef(new THREE.Vector3());
  1162→  const targetLookAt = useRef(new THREE.Vector3(0, buildingHeight * 0.3, 0));
  1163→  const animFrames = useRef(0);
  1164→  const lerpFactor = 0.12;
  1165→
  1166→  // Recompute target whenever inputs change.
  1167→  useEffect(() => {
  1168→    const maxDim = Math.max(plotWidth, plotLength, 20);
  1169→    const h = Math.max(buildingHeight, maxDim * 0.4);
  1170→    switch (cameraView) {
  1171→      case 'top':
  1172→        targetPos.current.set(0.01, maxDim * 1.55, 0.01);
  1173→        targetLookAt.current.set(0, 0, 0);
  1174→        break;
  1175→      case 'front':
  1176→        targetPos.current.set(0, h * 0.55, maxDim * 1.25);
  1177→        targetLookAt.current.set(0, h * 0.45, 0);
  1178→        break;
  1179→      case 'isometric':
  1180→        targetPos.current.set(
  1181→          maxDim * 0.95,
  1182→          maxDim * 1.0,
  1183→          maxDim * 0.95,
  1184→        );
  1185→        targetLookAt.current.set(0, h * 0.35, 0);
  1186→        break;
  1187→      case 'orbit':
  1188→      default:
  1189→        targetPos.current.set(
  1190→          maxDim * 0.8,
  1191→          h + maxDim * 0.45,
  1192→          maxDim * 1.05,
  1193→        );
  1194→        targetLookAt.current.set(0, h * 0.3, 0);
  1195→        break;
  1196→    }
  1197→    animFrames.current = 75; // ~1.25s of animation at 60fps
  1198→  }, [cameraView, plotWidth, plotLength, buildingHeight]);
  1199→
  1200→  // Cancel animation when user starts dragging.
  1201→  useEffect(() => {
  1202→    const c = controls as { addEventListener?: (e: string, fn: () => void) => void; removeEventListener?: (e: string, fn: () => void) => void } | null;
  1203→    if (!c || !c.addEventListener) return;
  1204→    const onStart = () => {
  1205→      animFrames.current = 0;
  1206→    };
  1207→    c.addEventListener('start', onStart);
  1208→    return () => {
  1209→      c.removeEventListener?.('start', onStart);
  1210→    };
  1211→  }, [controls]);
  1212→
  1213→  useFrame(() => {
  1214→    if (animFrames.current <= 0) return;
  1215→    animFrames.current -= 1;
  1216→    camera.position.lerp(targetPos.current, lerpFactor);
  1217→    const c = controls as { target?: THREE.Vector3; update?: () => void } | null;
  1218→    if (c?.target && c?.update) {
  1219→      c.target.lerp(targetLookAt.current, lerpFactor);
  1220→      c.update();
  1221→    } else {
  1222→      camera.lookAt(targetLookAt.current);
  1223→    }
  1224→  });
  1225→
  1226→  return null;
  1227→}
